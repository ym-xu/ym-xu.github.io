{{- /* The writing habit: one square per day for the last 26 weeks, darker for more pieces (essays and bookmark takes both count), plus the weekly streak. Built at deploy time; the daily deploy keeps "today" current. */ -}} {{- $pages := where site.RegularPages "Section" "in" (slice "posts" "bookmarks") -}} {{- $count := dict -}} {{- range $pages -}} {{- $k := .Date.Format "2006-01-02" -}} {{- $count = merge $count (dict $k (add (index $count $k | default 0) 1)) -}} {{- end -}} {{- $weeks := 26 -}} {{- $wd := printf "%d" now.Weekday | int -}}{{- /* Sunday = 0 */ -}} {{- $back := add (mod (add $wd 6) 7) (mul 7 (sub $weeks 1)) -}}{{- /* back to a Monday */ -}} {{- $start := now.AddDate 0 0 (sub 0 $back) -}} {{- $total := 0 -}}{{- $active := slice -}}
{{- range $w := seq 0 (sub $weeks 1) -}} {{- $n := 0 -}}
{{- range $d := seq 0 6 -}} {{- $day := $start.AddDate 0 0 (add (mul $w 7) $d) -}} {{- if le $day.Unix now.Unix -}} {{- $c := index $count ($day.Format "2006-01-02") | default 0 -}} {{- $n = add $n $c -}} {{- else -}}{{- end -}} {{- end -}}
{{- $total = add $total $n -}} {{- $active = $active | append (gt $n 0) -}} {{- end -}}
{{- /* weekly streak: full weeks with something in them, counting back; this week counts once it has a piece, and doesn't break the streak while it's empty */ -}} {{- $streak := 0 -}}{{- $going := true -}} {{- range $i := seq (sub $weeks 1) -1 0 -}} {{- $on := index $active $i -}} {{- if and $going $on -}}{{- $streak = add $streak 1 -}} {{- else if and $going (ne $i (sub $weeks 1)) -}}{{- $going = false -}}{{- end -}} {{- end -}}

{{ $total }} piece{{ if ne $total 1 }}s{{ end }} in 26 weeks · {{ $streak }}-week streak · {{ if index $active (sub $weeks 1) }}written this week{{ else }}nothing yet this week{{ end }}
