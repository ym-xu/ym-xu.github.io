(lower $tag) }}{{ $hint = printf "Write it as %q." . }}{{ end }}{{ end -}} {{- errorf "%s: tag %q is not in data/tags.yaml. %s" $page.File.Path $tag $hint -}} {{- end -}} {{- end -}} {{- end -}}
