{{- /* Previous / next among a page's siblings, in the order they are given. Used where the order is by weight: PrevInSection walks by date, which runs backwards for weighted pages. Usage: partial "pager-siblings.html" (dict "page" . "sibs" $sibs "label" "More projects") label (optional) becomes the nav's aria-label. */ -}} {{- $page := .page -}} {{- $sibs := .sibs -}} {{- $prev := false -}} {{- $next := false -}} {{- range $i, $s := $sibs -}} {{- if eq $s.RelPermalink $page.RelPermalink -}} {{- if gt $i 0 }}{{ $prev = index $sibs (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $sibs) }}{{ $next = index $sibs (add $i 1) }}{{ end -}} {{- end -}} {{- end -}}
{{ with $prev }}← {{ .Title }}{{ end }} {{ with $next }}{{ .Title }} →{{ end }}
